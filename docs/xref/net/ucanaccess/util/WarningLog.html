1   package net.ucanaccess.util;
2   
3   import java.lang.System.Logger;
4   import java.lang.System.Logger.Level;
5   import java.text.MessageFormat;
6   import java.util.LinkedHashMap;
7   import java.util.Map;
8   
9   /**
10   * Logs warnings without flooding the log when the same kind of warning comes up over and over.
11   * <p>
12   * Only the first few warnings of each kind are logged at level {@code WARNING}, further ones go to level
13   * {@code DEBUG}. The kind of a warning is its message pattern. {@link #summarize()} reports how many warnings were
14   * moved to {@code DEBUG}, typically at the end of a longer operation.
15   * <p>
16   * An example is loading a damaged database, which can cause the same warning for many tables, columns or rows.
17   * <p>
18   * Instances are not thread-safe.
19   *
20   * @since 5.3.0
21   */
22  public final class WarningLog {
23      private static final int DEFAULT_LIMIT = 5;
24  
25      private final Logger            logger;
26      private final int               limit;
27      private final Map<String, Kind> kinds = new LinkedHashMap<>();
28  
29      /**
30       * Creates a warning log with the default limit.
31       *
32       * @param logger the logger to write to
33       */
34      public WarningLog(Logger logger) {
35          this(logger, DEFAULT_LIMIT);
36      }
37  
38      /**
39       * Creates a warning log.
40       *
41       * @param logger the logger to write to
42       * @param limit  the number of warnings of each kind logged at level {@code WARNING}
43       */
44      public WarningLog(Logger logger, int limit) {
45          this.logger = logger;
46          this.limit = limit;
47      }
48  
49      /**
50       * Logs a warning.
51       * <p>
52       * The pattern must be a constant, the variable parts belong into {@code params}. A pattern assembled at
53       * runtime would make every warning a kind of its own.
54       *
55       * @param pattern the message pattern in {@link MessageFormat} syntax, identifies the kind of warning
56       * @param params  the message parameters
57       */
58      public void warn(String pattern, Object... params) {
59          warn(null, pattern, params);
60      }
61  
62      /**
63       * Logs a warning caused by an exception. The stack trace of the exception is only logged if level
64       * {@code DEBUG} is enabled. As with {@link #warn(String, Object...)}, the pattern must be a constant.
65       *
66       * @param cause   the exception, may be {@code null}
67       * @param pattern the message pattern in {@link MessageFormat} syntax, identifies the kind of warning
68       * @param params  the message parameters
69       */
70      public void warn(Throwable cause, String pattern, Object... params) {
71          String message = MessageFormat.format(pattern, params);
72          Kind kind = kinds.computeIfAbsent(pattern, p -> new Kind(message));
73          kind.count++;
74          Level level = kind.count <= limit ? Level.WARNING : Level.DEBUG;
75          if (cause != null && logger.isLoggable(Level.DEBUG)) {
76              logger.log(level, message, cause);
77          } else {
78              logger.log(level, message);
79          }
80      }
81  
82      /**
83       * Logs one warning for each kind that exceeded the limit, then starts counting anew.
84       */
85      public void summarize() {
86          kinds.values().stream()
87              .filter(k -> k.count > limit)
88              .forEach(k -> logger.log(Level.WARNING, MessageFormat.format(
89                  "{0} more warnings like \"{1}\" were logged at level DEBUG", k.count - limit, k.first)));
90          kinds.clear();
91      }
92  
93      /**
94       * Counts the warnings of one kind and keeps the first message for the summary.
95       */
96      private static final class Kind {
97          private final String first;
98          private int          count;
99  
100         /**
101          * Creates a kind of warning.
102          *
103          * @param first the first message of this kind
104          */
105         private Kind(String first) {
106             this.first = first;
107         }
108     }
109 
110 }
