38      private static boolean hasPassword(File f) throws IOException {
39          try (Database db = Try.catching(() -> DatabaseBuilder.open(f))
40              .orElseGet(() -> Try.catching(() -> new DatabaseBuilder()
41                  .withReadOnly(true)
42                  .withFile(f).open()).orThrow())) {
43              return db.getDatabasePassword() != null;
44          }
45      }

230     private void start(String[] commands) {
231         StringBuilder sb = new StringBuilder();
232         boolean exit = false;
233         while (connected) {
234             String userInput;
235             if (commands != null) {
236                 userInput = String.join(" ", commands);
237                 if (!userInput.endsWith(";")) {
238                    userInput += ";";
239                 }
240                 commands = null;
241                 exit = true;
242             } else {
243                 userInput = readInput();
244             }
245             if ("quit".equalsIgnoreCase(userInput)) {
246                 connected = false;
247                 break;
248             }
249             sb.append(" ").append(userInput);
250 
251             // If the current userInput ends with ';', then execute the buffered command.
252             if (userInput.endsWith(";")) {
253                 String cmd = sb.substring(0, sb.length() - 1).trim();
254                 try {
255                     if (cmd.toLowerCase(Locale.ROOT).startsWith("export ")) {
256                         executeExport(cmd);
257                     } else {
258                         executeStatement(cmd);
259                     }
260                 } catch (Exception ex) {
261                     prompt(ex.getMessage());
262                 }
263                 if (exit) {
264                    connected = false;
265                    break;
266                 }
267                 sb.setLength(0);
268                 prompt();
269             }
270         }
271         System.out.println("Cheers! Thank you for using the UCanAccess JDBC Driver.");
272     }

274     /**
275      * Parse the {@code cmd} to handle command line flags of the form: "export [-d delimiter] [-t table] pathToCsv". For
276      * example:
277      *
278      * <pre>
279      * export -d , -t License License.csv
280      * </pre>
281      *
282      * The {@code -d ,} option changes the delimiter character to a comma instead of the default semicolon. The
283      * {@code -t License} option dumps the {@code License} table using the SQL statement "SELECT * FROM [License]".
284      */
285     private void executeExport(String cmd) throws SQLException, IOException {
286         List<String> tokens = tokenize(cmd);
287 
288         Exporter.Builder exporterBuilder = new Exporter.Builder();
289         String table = null;
290         String schemaFileName = null;
291 
292         // Process the command line flags.
293         int i = 1; // skip the first token which will always be "export"
294         for (; i < tokens.size(); i++) {
295             String arg = tokens.get(i);
296             if (!arg.startsWith("-")) {
297                 break;
298             }
299             switch (arg) {
300                 case "-d":
301                     i++;
302                     if (i >= tokens.size()) {
303                         prompt("Missing parameter for -d flag");
304                         prompt(EXPORT_PROMPT);
305                         return;
306                     }
307                     exporterBuilder.withDelimiter(tokens.get(i));
308                     break;
309                 case "-t":
310                     i++;
311                     if (i >= tokens.size()) {
312                         prompt("Missing parameter for -t flag");
313                         prompt(EXPORT_PROMPT);
314                         return;
315                     }
316                     table = tokens.get(i);
317                     break;
318                 case "--bom":
319                     exporterBuilder.includeBom(true);
320                     break;
321                 case "--newlines":
322                     exporterBuilder.preserveNewlines(true);
323                     break;
324                 case "--big_query_schema":
325                     i++;
326                     if (i >= tokens.size()) {
327                         prompt("Missing parameter for --big_query_schema flag");
328                         prompt(EXPORT_PROMPT);
329                         return;
330                     }
331                     schemaFileName = tokens.get(i);
332                     break;
333                 case "--help":
334                     printExportHelp();
335                     return;
336                 case "--":
337                     i++;
338                     break;
339                 default:
340                     prompt("Unknown flag " + arg);
341                     prompt(EXPORT_PROMPT);
342                     return;
343             }
344 
345         }
346         if (i >= tokens.size()) {
347             prompt("File name not found");
348             prompt(EXPORT_PROMPT);
349             return;
350         }
351         if (i < tokens.size() - 1) {
352             prompt("Too many arguments");
353             prompt(EXPORT_PROMPT);
354             return;
355         }
356         String csvFileName = tokens.get(i);
357         Exporter exporter = exporterBuilder.build();
358 
359         // Determine the SQL statement to execute. If the '-t table' option is given
360         // run a 'select * from [table]' query to export the table, instead of
361         // executing the 'lastSqlQuery'.
362         String sqlQuery;
363         if (table != null && !table.isEmpty()) {
364             sqlQuery = Sql.of(SELECT, "*", FROM, "[" + table + "]").toString();
365         } else if (lastSqlQuery != null) {
366             sqlQuery = lastSqlQuery;
367         } else {
368             prompt("You must first execute an SQL query, then export the ResultSet!");
369             return;
370         }
371 
372         exportCsvAndSchema(sqlQuery, csvFileName, schemaFileName, exporter);
373     }

375     /**
376      * Runs the given{@code sqlQuery} and exports the CSV to {@code csvFileName}. and the BigQuery schema to
377      * {@code schemaFileName} (if given). Uses the given {@code exporter} to perform the actual CSV and schema export
378      * operations.
379      */
380     private void exportCsvAndSchema(CharSequence sqlQuery, String csvFileName, String schemaFileName, Exporter exporter) throws SQLException, IOException {
381         Objects.requireNonNull(sqlQuery, "Sql required");
382         try (Statement statement = conn.createStatement()) {
383             ResultSet rs = statement.executeQuery(sqlQuery.toString());
384 
385             // output the csvFile
386             File csvFile = new File(csvFileName);
387             try (PrintStream out = new PrintStream(csvFile)) {
388                 exporter.dumpCsv(rs, out);
389                 out.flush();
390             }
391             prompt("Created CSV file: " + csvFile.getAbsolutePath());
392 
393             // output the schema file if requested
394             if (schemaFileName != null) {
395                 File schemaFile = new File(schemaFileName);
396                 try (PrintStream out = new PrintStream(csvFile)) {
397                     exporter.dumpSchema(rs, out);
398                     out.flush();
399                 }
400                 prompt("Created schema file: " + schemaFile.getAbsolutePath());
401             }
402         }
403     }

405     static List<String> tokenize(String s) throws IOException {
406         // Create a tokenizer that creates tokens delimited by whitespace. Also handles single and
407         // double quotes. Does not support comments or numbers. All characters between U+0000
408         // U+0020 are considered to be whitespace characters (this includes \r, \n, and \t).
409         // All characters from U+0021 to U+00FF are normal word characters.
410         // Internally, StreamTokenzier treats all characters >= U+0100 to be a normal
411         // word character, and unfortunately, this cannot be changed.
412         //
413         // Non-breaking-space character U+00A0 is currently treated as a normal character,
414         // but I could be convinced that it should be considered a whitespace.
415         StreamTokenizer st = new StreamTokenizer(new StringReader(s));
416         st.resetSyntax();
417         st.wordChars(33, 255);
418         st.whitespaceChars(0, ' ');
419         st.quoteChar('"');
420         st.quoteChar('\'');
421 
422         List<String> tokens = new ArrayList<>();
423         while (true) {
424             int ttype = st.nextToken();
425             if (ttype == StreamTokenizer.TT_EOF) {
426                 break;
427             } else if (ttype == StreamTokenizer.TT_WORD) {
428                 tokens.add(st.sval);
429             } else if (ttype == '\'' || ttype == '"') {
430                 tokens.add(st.sval);
431             }
432         }
433         return tokens;
434     }

440     static class TableFormat {
441         static final List<Integer>       NUMERIC_JDBC_TYPES =
442             List.of(Types.BIT, Types.TINYINT, Types.SMALLINT, Types.INTEGER, Types.BIGINT, Types.FLOAT,
443                 Types.REAL, Types.DOUBLE, Types.NUMERIC, Types.DECIMAL, Types.ROWID);
444         private static final int         MAX_COL_WIDTH      = 50;
445         // the table is indented to set it off from the text before
446         private static final String      INDENT             = "  ";
447         // wider tables are printed with one block of column names and values per row
448         private static final int         MAX_TABLE_WIDTH    = 200;
449         // code points displayed in two console columns, after the ranges of Markus Kuhn's wcwidth
450         private static final int[][]     WIDE_RANGES        = {
451             {0x1100, 0x115F}, // Hangul Jamo
452             {0x2E80, 0xA4CF}, // CJK radicals ... Yi
453             {0xAC00, 0xD7A3}, // Hangul syllables
454             {0xF900, 0xFAFF}, // CJK compatibility ideographs
455             {0xFE30, 0xFE4F}, // CJK compatibility forms
456             {0xFF00, 0xFF60}, // fullwidth forms
457             {0xFFE0, 0xFFE6}, // fullwidth signs
458             {0x1F300, 0x1F64F}, // emoji
459             {0x1F900, 0x1F9FF}, // supplemental symbols and pictographs
460             {0x20000, 0x3FFFD} // CJK extensions
461         };

463         private final Border             border;
464         private final List<String>       colNames;
465         private final List<Integer>      colWidths;
466         private final List<Integer>      colTypes;
467         private final List<List<String>> records            = new ArrayList<>();
468         private boolean                  hideEmptyColumns;
469         private int                      maxTableWidth      = MAX_TABLE_WIDTH;

471         /**
472          * Characters to draw the table border with.
473          */
474         enum Border {
475             ASCII('|', "+-++", "+=++", "+-++", "+-++"),
476             UNICODE('│', "┌─┬┐", "╞═╪╡", "├─┼┤", "└─┴┘");
477 
478             private final char   vertical;
479             // each line given by its left junction, horizontal character, inner and right junction
480             private final String top;
481             private final String header;
482             private final String middle;
483             private final String bottom;

485             /**
486              * Creates a border of the given characters.
487              *
488              * @param vertical  character of the vertical lines
489              * @param top       characters of the top line
490              * @param header    characters of the line below the header
491              * @param middle    characters of the line between two rows
492              * @param bottom    characters of the bottom line
493              */
494             Border(char vertical, String top, String header, String middle, String bottom) {
495                 this.vertical = vertical;
496                 this.top = top;
497                 this.header = header;
498                 this.middle = middle;
499                 this.bottom = bottom;

503              * Returns the unicode border if the given charset can encode its characters, the ascii border otherwise.
504              */
505             static Border forCharset(Charset charset) {
506                 String characters = UNICODE.vertical + UNICODE.top + UNICODE.header + UNICODE.middle + UNICODE.bottom;
507                 return charset.canEncode() && charset.newEncoder().canEncode(characters) ? UNICODE : ASCII;
508             }

517             /**
518              * Returns the charset of the standard output, as far as the running Java version tells it.
519              */
520             private static Charset consoleCharset() {
521                 // stdout.encoding exists since Java 19, sun.stdout.encoding on Windows consoles before, native.encoding
522                 // since Java 17 is what System.out uses on Java 18 when not writing to a console, sun.jnu.encoding is the
523                 // platform encoding on older versions, unlike the default charset not changed by -Dfile.encoding
524                 for (String property : new String[] {"stdout.encoding", "sun.stdout.encoding", "native.encoding", "sun.jnu.encoding"}) {
525                     String name = System.getProperty(property);
526                     try {
527                         if (name != null && Charset.isSupported(name)) {
528                             return Charset.forName(name);
529                         }
530                     } catch (IllegalArgumentException ignored) {
531                         // an illegal charset name, try the next property
532                     }
533                 }
534                 return Charset.defaultCharset();
535             }

537             /**
538              * Returns a horizontal line of the table.
539              *
540              * @param chars   left junction, horizontal character, inner and right junction of the line
541              * @param widths  widths of the column contents
542              */
543             static String line(String chars, List<Integer> widths) {
544                 StringJoiner line = new StringJoiner(chars.substring(2, 3), chars.substring(0, 1), chars.substring(3, 4));
545                 // one blank on either side of the content
546                 widths.forEach(w -> line.add(chars.substring(1, 2).repeat(w + 2)));
547                 return line.toString();
548             }
549 
550             /**
551              * Returns a horizontal line of the table with the given label at its start.
552              *
553              * @param chars   left junction, horizontal character, inner and right junction of the line
554              * @param widths  widths of the column contents, the first one wide enough for the label
555              * @param label   text to show in the line
556              */
557             static String line(String chars, List<Integer> widths, String label) {
558                 String line = line(chars, widths);
559                 String text = chars.charAt(1) + " " + label + " ";
560                 return line.charAt(0) + text + line.substring(1 + text.length());
561             }
562         }

573         /**
574          * Reads the rows of the given result set.
575          *
576          * @param resultSet  result set to format
577          * @param maxRows    maximum number of rows to read, negative for all rows
578          * @param border     characters to draw the table border with
579          */
580         TableFormat(ResultSet resultSet, int maxRows, Border border) throws SQLException {
581             this.border = border;
582             ResultSetMetaData meta = resultSet.getMetaData();
583             int columnCount = meta.getColumnCount();
584 
585             colNames = new ArrayList<>(columnCount);
586             colWidths = new ArrayList<>(columnCount);
587             colTypes = new ArrayList<>(columnCount);
588             for (int col = 1; col <= columnCount; ++col) {
589                 String colLabel = printable(meta.getColumnLabel(col));
590                 colNames.add(colLabel);
591                 colWidths.add(Math.min(displayWidth(colLabel), MAX_COL_WIDTH));
592                 colTypes.add(meta.getColumnType(col));
593             }
594 
595             records.add(colNames); // header record
596             while ((maxRows < 0 || maxRows >= records.size()) && resultSet.next()) {
597                 List<String> rec = new ArrayList<>();
598                 records.add(rec);
599                 for (int col = 1; col <= columnCount; ++col) {
600                     Object obj = resultSet.getObject(col);
601                     if (obj != null && obj.getClass().isArray() && !obj.getClass().getComponentType().isPrimitive()) {
602                         obj = Arrays.toString((Object[]) obj);
603                     }
604 
605                     String s = obj == null ? "" : printable(obj.toString());
606                     rec.add(s);
607 
608                     int width = Math.min(displayWidth(s), MAX_COL_WIDTH);
609                     if (colWidths.get(col - 1) < width) {
610                         colWidths.set(col - 1, width);
611                     }
612                 }
613             }
614         }

636         /**
637          * Prints the table to the given stream, followed by the number of rows and the hidden columns.
638          *
639          * @param printStream  stream to print to
640          */
641         void output(PrintStream printStream) {
642             List<Integer> columns = visibleColumns();
643             List<Integer> widths = columns.stream().map(colWidths::get).collect(Collectors.toList());
644             List<List<String>> rows = records.subList(1, records.size()); // index 0 holds header
645 
646             printStream.println();
647             // each column takes its width plus one blank on either side and a vertical line
648             if (widths.stream().mapToInt(w -> w + 3).sum() + 1 > maxTableWidth) {
649                 if (rows.isEmpty()) {
650                     printStream.println(INDENT + String.format("(0 rows, columns: %s)", String.join(", ", colNames)));
651                 } else {
652                     outputVertical(printStream, rows, columns);
653                 }
654                 return;
655             }
656             printStream.println(INDENT + Border.line(border.top, widths));
657             printStream.println(INDENT + row(colNames, columns));
658             if (!rows.isEmpty()) {
659                 printStream.println(INDENT + Border.line(border.header, widths));
660             }
661             rows.forEach(r -> printStream.println(INDENT + row(r, columns)));
662             printStream.println(INDENT + Border.line(border.bottom, widths));
663             printStream.println(INDENT + footer(rows.size(), columns));
664         }

666         /**
667          * Prints one block per row, listing the column names and values of the row line by line.
668          *
669          * @param printStream  stream to print to
670          * @param rows         rows to print
671          * @param columns      indexes of the columns to print
672          */
673         void outputVertical(PrintStream printStream, List<List<String>> rows, List<Integer> columns) {
674             // the name column is wide enough for the row label in the line above each block
675             String lastLabel = String.format("Row %d", rows.size());
676             int nameWidth = Math.min(MAX_COL_WIDTH, columns.stream()
677                 .mapToInt(col -> displayWidth(colNames.get(col)))
678                 .reduce(lastLabel.length() + 1, Math::max));
679             int valueWidth = Math.min(MAX_COL_WIDTH, rows.stream()
680                 .flatMap(row -> columns.stream().map(row::get))
681                 .mapToInt(TableFormat::displayWidth)
682                 .max().orElse(0));
683             List<Integer> widths = List.of(nameWidth, valueWidth);
684 
685             for (int idx = 0; idx < rows.size(); idx++) {
686                 String label = String.format("Row %d", idx + 1);
687                 printStream.println(INDENT + Border.line(idx == 0 ? border.top : border.middle, widths, label));
688                 for (int col : columns) {
689                     printStream.println(INDENT + cells(List.of(fit(colNames.get(col), nameWidth, false),
690                         fit(rows.get(idx).get(col), valueWidth, false))));
691                 }
692             }
693             printStream.println(INDENT + Border.line(border.bottom, widths));
694             printStream.println(INDENT + footer(rows.size(), columns));
695         }

697         /**
698          * Returns the indexes of the columns to print, which are all columns unless empty columns are hidden. All
699          * columns are printed if every one is empty.
700          */
701         List<Integer> visibleColumns() {
702             List<Integer> all = IntStream.range(0, colNames.size()).boxed().collect(Collectors.toList());
703             if (!hideEmptyColumns) {
704                 return all;
705             }
706             List<Integer> filled = all.stream()
707                 .filter(col -> records.stream().skip(1).anyMatch(rec -> !rec.get(col).isEmpty()))
708                 .collect(Collectors.toList());
709             return filled.isEmpty() ? all : filled;
710         }

712         /**
713          * Returns the line below the table with the number of rows and the names of the hidden columns.
714          *
715          * @param rowCount  number of rows printed
716          * @param columns   indexes of the columns printed
717          */
718         String footer(int rowCount, List<Integer> columns) {
719             StringJoiner footer = new StringJoiner(", ", "(", ")");
720             footer.add(rowCount == 1 ? "1 row" : String.format("%d rows", rowCount));
721             List<String> hidden = IntStream.range(0, colNames.size())
722                 .filter(col -> !columns.contains(col))
723                 .mapToObj(colNames::get)
724                 .collect(Collectors.toList());
725             if (!hidden.isEmpty()) {
726                 footer.add(String.format("%d empty %s hidden: %s", hidden.size(), hidden.size() == 1 ? "column" : "columns",
727                     String.join(", ", hidden)));
728             }
729             return footer.toString();
730         }

761         /**
762          * Replaces control characters such as line breaks and tabs and the unicode line and paragraph separators with a
763          * blank, as they would break the table.
764          */
765         static String printable(String str) {
766             return str.codePoints()
767                 .map(cp -> Character.isISOControl(cp) || Character.getType(cp) == Character.LINE_SEPARATOR
768                     || Character.getType(cp) == Character.PARAGRAPH_SEPARATOR ? ' ' : cp)
769                 .collect(StringBuilder::new, StringBuilder::appendCodePoint, StringBuilder::append)
770                 .toString();
771         }

773         /**
774          * Returns the number of console columns the given string takes up.
775          */
776         static int displayWidth(String str) {
777             return str.codePoints().map(TableFormat::displayWidth).sum();
778         }
779 
780         /**
781          * Returns the number of console columns the given code point takes up: 0 for combining marks, 2 for wide East
782          * Asian characters and emoji, 1 otherwise.
783          */
784         static int displayWidth(int codePoint) {
785             int type = Character.getType(codePoint);
786             if (type == Character.NON_SPACING_MARK || type == Character.ENCLOSING_MARK || type == Character.FORMAT) {
787                 return 0;
788             }
789             return isWide(codePoint) ? 2 : 1;
790         }
791 
792         /**
793          * Returns whether the given code point is displayed in two console columns.
794          */
795         static boolean isWide(int cp) {
796             if (cp == 0x303F) { // ideographic half fill space
797                 return false;
798             }
799             for (int[] range : WIDE_RANGES) {
800                 if (cp >= range[0] && cp <= range[1]) {
801                     return true;
802                 }
803             }
804             return false;
805         }

807         /**
808          * Pads or cuts the given string to exactly the given display width.
809          *
810          * @param str         string to fit
811          * @param width       display width of the result
812          * @param alignRight  whether to pad on the left
813          */
814         static String fit(String str, int width, boolean alignRight) {
815             StringBuilder sb = new StringBuilder();
816             int used = 0;
817             for (int cp : str.codePoints().toArray()) {
818                 int w = displayWidth(cp);
819                 if (used + w > width) {
820                     break;
821                 }
822                 sb.appendCodePoint(cp);
823                 used += w;
824             }
825             String padding = " ".repeat(width - used);
826             return alignRight ? padding + sb : sb + padding;
827         }
