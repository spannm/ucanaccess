13  public class BlobKey implements Serializable {
14      private static final long   serialVersionUID = -8580858159403159903L;
15      public static final int     MAX_SIZE         = 4096;
16  
17      // declared as Map rather than a concrete Serializable type; callers are expected to pass a serializable Map
18      // (this class is only ever populated with a HashMap internally, see the Table/Row constructor below)
19      @SuppressWarnings({"serial", "java:S1948"})
20      private Map<String, Object> key;
21      private final String        tableName;
22      private final String        columnName;
23  
24      public BlobKey(Map<String, Object> key, String tableName, String columnName) {
25          this.key = key;
26          this.tableName = tableName;
27          this.columnName = columnName;
28      }
29  
30      public BlobKey(Table table, String columnName, Row row) {
31          tableName = table.getName();
32          this.columnName = columnName;
33          if (hasPrimaryKey(table)) {
34              List<? extends Index.Column> cl = table.getPrimaryKeyIndex().getColumns();
35              Map<String, Object> keyMap = new HashMap<>();
36              for (Index.Column c : cl) {
37                  keyMap.put(c.getName(), row.get(c.getName()));
38              }
39              key = keyMap;
40          }
41      }
42  
43      public static boolean hasPrimaryKey(Table table) {
44          for (Index idx : table.getIndexes()) {
45              if (idx.isPrimaryKey()) {
46                  return true;
47              }
48          }
49          return false;
50      }
51  
52      public OleBlob getOleBlob(Database db) throws UcanaccessSQLException {
53          return Try.catching(() -> {
54              Table t = db.getTable(tableName);
55              Cursor c = CursorBuilder.createPrimaryKeyCursor(t);
56              return c.findFirstRow(key) ? c.getCurrentRow().getBlob(columnName) : null;
57          }).orThrow(UcanaccessSQLException::new);
58      }
59  
60      public byte[] getBytes() throws UcanaccessSQLException {
61          return Try.withResources(ByteArrayOutputStream::new, bais -> {
62              ObjectOutputStream oos = new ObjectOutputStream(bais);
63              oos.writeObject(this);
64              oos.flush();
65              return bais.toByteArray();
66          }).orThrow(UcanaccessSQLException::new);
67      }
68  
69      public static BlobKey getBlobKey(byte[] bytes) {
70          return Try.withResources(() -> new ByteArrayInputStream(bytes), bais -> {
71              ObjectInputStream ois = new ObjectInputStream(bais);
72              Object obj = ois.readObject();
73              return obj instanceof BlobKey ? (BlobKey) obj : null;
74          }).orIgnore();
75      }
76  
77      public static BlobKey getBlobKey(InputStream is) {
78          return Try.catching(() -> {
79              byte[] bt = new byte[MAX_SIZE];
80              int readBytes = is.read(bt);
81              return readBytes > 0 ? getBlobKey(bt) : null;
82          }).orIgnore();
83      }
84  
85      public Map<String, Object> getKey() {
86          return key;
87      }
88  
89      public String getColumnName() {
90          return columnName;
91      }
92  
93  }
