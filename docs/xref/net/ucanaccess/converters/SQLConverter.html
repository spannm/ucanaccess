144     private static void aliases(String sql, NormalizedSQL nsql) {
145         Matcher m = Patterns.SELECT_FROM_START.matcher(sql);
146         if (m.find()) {
147 
148             int init = m.end();
149             sql = sql.substring(init);
150             m = Patterns.SELECT_FROM_END.matcher(sql);
151             if (m.find()) {
152                 int end = m.start();
153                 sql = sql.substring(0, end);
154 
155                 for (m = Patterns.UNESCAPED_ALIAS.matcher(sql); m.find(); m = Patterns.UNESCAPED_ALIAS.matcher(sql)) {
156                     int e = m.end();
157 
158                     sql = sql.substring(e) + " ";
159                     char[] sqlc = sql.toCharArray();
160                     if (sqlc[0] == '[') {
161                         continue;
162                     }
163                     StringBuilder sb = new StringBuilder();
164                     for (char c : sqlc) {
165                         if (c == ' ' || c == '\n' || c == '\r' || c == ',') {
166                             String key = preEscapingIdentifier(sb.toString());
167                             nsql.put(key, sb.toString());
168                             break;
169                         } else {
170                             sb.append(c);
171                         }
172                     }
173                 }
174 
175             }
176         }
177     }

180         Matcher m1 = Patterns.SELECT_IDENTITY.matcher(sql);
181         Matcher m2 = Patterns.HAS_FROM.matcher(sql);
182         String end = m1.matches() && !m2.find() ? " FROM DUAL" : "";
183         if (key instanceof String) {

185         }
186         return Patterns.IDENTITY.matcher(sql).replaceAll("$1" + key + "$3") + end;

193     private static int[] getQuoteGroup(String s) {
194         if (!s.contains("''")) {
195             Matcher m = Patterns.QUOTE_M.matcher(s);
196             if (m.find()) {
197                 return new int[] {m.start(), m.end()};
198             }
199         } else {
200             int[] ret = new int[] {-1, -1};
201             Matcher m = Patterns.QUOTE_S.matcher(s);
202             while (m.find()) {
203                 int start = m.start();
204                 int end = m.end();
205                 if ((end - start) % 2 == 0) {
206                     if (ret[0] == -1) {
207                         return new int[] {m.start(), m.end()};
208                     }
209                 } else {
210                     if (ret[0] == -1) {
211                         ret[0] = m.start();
212                     } else {
213                         ret[1] = m.end();
214                         return ret;
215                     }
216                 }
217             }
218         }
219         return new int[0];
220     }

222     private static int[] getDoubleQuoteGroup(String s) {
223         if (!s.contains("\"\"")) {
224             Matcher m = Patterns.DOUBLE_QUOTE_M.matcher(s);
225             if (m.find()) {
226                 return new int[] {m.start(), m.end()};
227             }
228         } else {
229             int[] ret = new int[] {-1, -1};
230             Matcher mc = Patterns.DOUBLE_QUOTE_S.matcher(s);
231             while (mc.find()) {
232                 int start = mc.start();
233                 int end = mc.end();
234                 if ((end - start) % 2 == 0) {
235                     if (ret[0] == -1) {
236                         return new int[] {mc.start(), mc.end()};
237                     }
238                 } else {
239                     if (ret[0] == -1) {
240                         ret[0] = mc.start();
241                     } else {
242                         ret[1] = mc.end();
243                         return ret;
244                     }
245                 }
246             }
247         }
248         return new int[0];

370     private static String replaceWorkAroundFunctions(String sql) {
371         sql = Patterns.DOLLAR_FUNCTIONS.matcher(sql).replaceAll("$1$2(");
372         for (String waFun : WORKAROUND_FUNCTIONS) {
373             sql = sql.replaceAll("(\\W)(?i)" + waFun + "\\s*\\(", "$1" + waFun + "WA(");
374         }
375         sql = sql.replaceAll("(\\W)(?i)STDEV\\s*\\(", "$1STDDEV_SAMP(");
376         sql = sql.replaceAll("(\\W)(?i)STDEVP\\s*\\(", "$1STDDEV_POP(");
377         sql = sql.replaceAll("(\\W)(?i)VAR\\s*\\(", "$1VAR_SAMP(");
378         sql = sql.replaceAll("(\\W)(?i)VARP\\s*\\(", "$1VAR_POP(");
379         return sql.replaceAll("(\\W)(?i)currentUser\\s*\\(", "$1user(");
380     }

382     public static String restoreWorkAroundFunctions(String sql) {

386         }
387         sql = sql.replaceAll("(\\W)(?i)STDDEV_SAMP\\s*\\(", "$1STDEV(");
388         sql = sql.replaceAll("(\\W)(?i)STDDEV_POP\\s*\\(", "$1STDEVP(");
389         sql = sql.replaceAll("(\\W)(?i)VAR_SAMP\\s*\\(", "$1VAR(");
390         sql = sql.replaceAll("(\\W)(?i)VAR_POP\\s*\\(", "$1VARP(");
391         return sql.replaceAll("(\\W)(?i)user\\s*\\(", "$1currentUser(");

409     public static NormalizedSQL convertSQL(String inputSql, UcanaccessConnection conn, boolean creatingQuery) {
410         NormalizedSQL nsql = new NormalizedSQL();
411         String sql = inputSql + " ";
412         aliases(sql, nsql);
413         sql = replaceBacktick(sql);
414         sql = convertUnion(sql);
415         sql = convertAccessDate(sql);
416         // quoted aliases first, so that apostrophised aliases are not mangled by replaceAposNames
417         sql = convertQuotedAliases(sql, nsql);
418         sql = replaceAposNames(sql);
419         sql = escape(sql);
420         sql = convertLike(sql);
421         sql = replaceWhiteSpacedTables(sql);
422         sql = translateAccessPowerOperators(sql);
423         sql = translateAccessIntegerOperators(sql);
424         if (!creatingQuery) {
425             Pivot.checkAndRefreshPivot(sql, conn);
426             sql = DFunction.convertDFunctions(sql, conn);
427         }
428         sql = sql.trim();
429 
430         nsql.setSql(sql);
431 
432         return nsql;
433     }

601     /**
602      * Translates the Access power operator ^ into calls of the function POWER. Like in Access, ^ binds stronger than a
603      * sign and chains from left to right, so -2^2 becomes -POWER(2, 2) and 2^3^2 becomes POWER(POWER(2, 3), 2).
604      * <p>
605      * Operands are found with regular expressions, so they cannot be nested deeper than one level of parentheses and
606      * cannot be subqueries.
607      *
608      * @param sql SQL string
609      * @return translated SQL string
610      */
611     static String translateAccessPowerOperators(String sql) {
612         if (sql == null || !sql.contains("^")) {
613             return sql;
614         }
615         return replaceOutsideTextLiterals(sql, part -> replaceRepeatedly(part, Patterns.POWER_OPERATION, "POWER($1, $2)"));
616     }

618     /**
619      * Translates the Access operators Mod and \\ (integer division) into calls of functions that round the operands
620      * to whole numbers first, like Access. As in Access, * and / bind stronger than \\, which binds stronger than Mod,
621      * so a * b Mod c \\ d becomes AccessMod(a * b, AccessIntDiv(c, d)). Operands follow the same rules as for the
622      * power operator.
623      *
624      * @param sql SQL string
625      * @return translated SQL string
626      */
627     static String translateAccessIntegerOperators(String sql) {
628         if (sql == null || (!sql.contains("\\") && !sql.toUpperCase(Locale.ROOT).contains("MOD"))) {
629             return sql;
630         }
631         return replaceOutsideTextLiterals(sql, part -> replaceRepeatedly(
632             replaceRepeatedly(part, Patterns.INTEGER_DIVISION, "AccessIntDiv($1, $2)"), Patterns.MOD_OPERATION, "AccessMod($1, $2)"));
633     }

635     /**
636      * Replaces all matches of a binary operation until none is left, so that in chains like a Mod b Mod c the call
637      * for the first operation becomes the left operand of the next one.
638      *
639      * @param sql         SQL string
640      * @param operation   pattern of the operation with both operands as groups
641      * @param replacement replacement referring to the operands as $1 and $2
642      * @return SQL string without matches
643      */
644     private static String replaceRepeatedly(String sql, Pattern operation, String replacement) {
645         String previous;
646         do {
647             previous = sql;
648             sql = operation.matcher(sql).replaceAll(replacement);
649         } while (!sql.equals(previous));
650         return sql;
651     }

658      * @return converted SQL string

769     private static String escape(String sql) {
770         int li = Math.max(sql.lastIndexOf('"'), sql.lastIndexOf('\''));
771         boolean enddq = sql.endsWith("\"") || sql.endsWith("'");
772         String suff = enddq ? "" : sql.substring(li + 1);
773         suff = convertPartIdentifiers(suff);
774         String tsql = enddq ? sql : sql.substring(0, li + 1);
775         int[] fd = getDoubleQuoteGroup(tsql);
776         int[] fs = getQuoteGroup(tsql);
777         if (fd.length > 0 || fs.length > 0) {
778             boolean inid = fs.length == 0 || (fd.length > 0 && fd[0] < fs[0]);
779             String group;
780             String str;
781             int[] mcr = inid ? fd : fs;
782             group = tsql.substring(mcr[0] + 1, mcr[1] - 1);
783             if (inid) {
784                 group = group.replace("'", "''").replace("\"\"", "\"");
785             }
786             str = tsql.substring(0, mcr[0]);
787             str = convertPartIdentifiers(str);
788             tsql = str + "'" + group + "'" + escape(tsql.substring(mcr[1]));
789         } else {
790             tsql = convertPartIdentifiers(tsql);
791         }
792         return tsql + suff;
793     }

996     static String convertToRegexMatches(String likeContent) {
997         if (likeContent == null || likeContent.isEmpty()) {
998             return likeContent;

1056     private static String convertDigit(String sql) {
1057         Matcher mtc = Patterns.ESPRESSION_DIGIT.matcher(sql);
1058         char[] cq = sql.toCharArray();
1059         if (mtc.find()) {
1060             int idx = mtc.start();
1061             int idxe = mtc.end();
1062             boolean replace = true;
1063             for (int j = idx; j >= 0; j--) {
1064                 if (!Character.isDigit(cq[j])) {
1065                     replace = !Character.isLetter(cq[j]);
1066                     break;
1067                 }
1068             }
1069             if (replace) {
1070                 return sql.substring(0, idx) + mtc.group(1) + "E0" + convertDigit(sql.substring(idxe));
1071             } else {
1072                 return sql.substring(0, idxe) + convertDigit(sql.substring(idxe));
1073             }
1074         } else {
1075             return sql;
1076         }

1080     private static String convertFormula0(String sql) {
1081         int li = Math.max(sql.lastIndexOf('\"'), sql.lastIndexOf('\''));
1082         boolean enddq = sql.endsWith("\"") || sql.endsWith("'");
1083         String suff = enddq ? "" : sql.substring(li + 1);
1084         suff = convertDigit(suff);
1085         String tsql = enddq ? sql : sql.substring(0, li + 1);
1086         int[] fd = getDoubleQuoteGroup(tsql);
1087         int[] fs = getQuoteGroup(tsql);
1088         if (fd.length > 0 || fs.length > 0) {
1089             boolean inid = fs.length == 0 || (fd.length > 0 && fd[0] < fs[0]);
1090             String group;
1091             String str;
1092             int[] mcr = inid ? fd : fs;
1093             group = tsql.substring(mcr[0], mcr[1]);
1094             str = tsql.substring(0, mcr[0]);
1095             str = convertDigit(str);
1096             tsql = str + group + convertFormula0(tsql.substring(mcr[1]));
1097         } else {
1098             tsql = convertDigit(tsql);
1099         }
1100         return tsql + suff;

1118     private static int secondOperandIndex(String sql, int i) {
1119         sql = sql.substring(i + 1);
1120         char[] ca = sql.toCharArray();
1121         boolean foundType = false;
1122         boolean field = false;
1123         boolean group = false;
1124         boolean digit = false;
1125         int countPar = 0;
1126         int j;
1127 
1128         for (j = 0; j < ca.length; j++) {
1129             char c = ca[j];
1130             if (c == ' ') {
1131                 continue;
1132             }
1133 
1134             if (foundType) {
1135                 if (field && c == ']') {
1136                     return j;
1137                 } else if (digit && !Character.isDigit(c) && c != '.') {
1138                     return j - 1;
1139                 } else if (group) {
1140                     if (c == '(') {
1141                         countPar++;
1142                     }
1143                     if (c == ')') {
1144                         countPar--;
1145                         if (countPar == 0) {
1146                             return j;
1147                         }
1148                     }
1149                 }
1150             } else {
1151                 if (c == '[') {
1152                     foundType = true;
1153                     field = true;
1154                 } else if (c == '(') {
1155                     foundType = true;
1156                     group = true;
1157                     countPar++;
1158                 } else if (Character.isDigit(c)
1159                     || ((c == '+' || c == '-') && j + 1 < ca.length && ca[j + 1] != '(' && ca[j + 1] != '[')) {
1160                     foundType = true;
1161                     digit = true;
1162                 }
1163 
1164             }
1165 
1166         }
1167         return j - 1;
