155     private static void aliases(String sql, NormalizedSQL nsql) {
156         Matcher m = Patterns.SELECT_FROM_START.matcher(sql);
157         if (m.find()) {
158 
159             int init = m.end();
160             sql = sql.substring(init);
161             m = Patterns.SELECT_FROM_END.matcher(sql);
162             if (m.find()) {
163                 int end = m.start();
164                 sql = sql.substring(0, end);
165 
166                 for (m = Patterns.UNESCAPED_ALIAS.matcher(sql); m.find(); m = Patterns.UNESCAPED_ALIAS.matcher(sql)) {
167                     int e = m.end();
168 
169                     sql = sql.substring(e) + " ";
170                     char[] sqlc = sql.toCharArray();
171                     if (sqlc[0] == '[') {
172                         continue;
173                     }
174                     StringBuilder sb = new StringBuilder();
175                     for (char c : sqlc) {
176                         if (c == ' ' || c == '\n' || c == '\r' || c == ',') {
177                             String key = preEscapingIdentifier(sb.toString());
178                             nsql.put(key, sb.toString());
179                             break;
180                         } else {
181                             sb.append(c);
182                         }
183                     }
184                 }
185 
186             }
187         }
188     }

204     private static int[] getQuoteGroup(String s) {
205         if (!s.contains("''")) {
206             Matcher m = Patterns.QUOTE_M.matcher(s);
207             if (m.find()) {
208                 return new int[] {m.start(), m.end()};
209             }
210         } else {
211             int[] ret = new int[] {-1, -1};
212             Matcher m = Patterns.QUOTE_S.matcher(s);
213             while (m.find()) {
214                 int start = m.start();
215                 int end = m.end();
216                 if ((end - start) % 2 == 0) {
217                     if (ret[0] == -1) {
218                         return new int[] {m.start(), m.end()};
219                     }
220                 } else {
221                     if (ret[0] == -1) {
222                         ret[0] = m.start();
223                     } else {
224                         ret[1] = m.end();
225                         return ret;
226                     }
227                 }
228             }
229         }
230         return new int[0];
231     }

420     public static NormalizedSQL convertSQL(String inputSql, UcanaccessConnection conn, boolean creatingQuery) {
421         NormalizedSQL nsql = new NormalizedSQL();
422         String sql = inputSql + " ";
423         aliases(sql, nsql);
424         sql = replaceBacktick(sql);
425         sql = convertUnion(sql);
426         sql = convertAccessDate(sql);
427         // quoted aliases first, so that apostrophised aliases are not mangled by replaceAposNames
428         sql = convertQuotedAliases(sql, nsql);
429         sql = replaceAposNames(sql);
430         sql = escape(sql);
431         sql = convertLike(sql);

632     /**
633      * Translates the Access concatenation operator &amp; into ||. Like in Access, &amp; binds weaker than + and -, so
634      * {@code 1 & 2 + 3} is "15" and {@code 1 + 2 & 3} is "33", and it converts numbers to text. HSQLDB gives || the
635      * same precedence as + and cannot concatenate two numbers, so each chain of operands joined by &amp; is put in
636      * parentheses and its first operand is cast to text: {@code a & b + c} becomes
637      * {@code CAST((a) AS LONGVARCHAR) || (b + c)}. An operand ends at a comma, a comparison, an unmatched parenthesis
638      * or a keyword like AND or AS.
639      *
640      * @param sql SQL string with text literals in single quotes and identifiers in double quotes
641      * @return translated SQL string
642      */
643     static String translateAccessConcatenation(String sql) {
644         if (sql == null || sql.indexOf('&') < 0) {
645             return sql;
646         }
647         String s = sql;
648         for (int amp = findConcatOperator(s, 0); amp >= 0; amp = findConcatOperator(s, 0)) {
649             boolean[] quoted = quotedPositions(s);
650             int start = concatOperandStart(s, quoted, amp);
651             List<String> operands = new ArrayList<>();
652             operands.add(s.substring(start, amp).trim());
653             int end = amp;
654             int next = amp;
655             while (next < s.length() && s.charAt(next) == '&' && !quoted[next]) {
656                 end = concatOperandEnd(s, quoted, next + 1);
657                 operands.add(s.substring(next + 1, end).trim());
658                 next = end;
659                 while (next < s.length() && Character.isWhitespace(s.charAt(next))) {
660                     next++;
661                 }
662             }
663             String replacement;
664             if (operands.stream().anyMatch(String::isEmpty)) {
665                 replacement = String.join(" || ", operands);
666             } else {
667                 replacement = "CAST((" + operands.get(0) + ") AS LONGVARCHAR)" + operands.stream().skip(1)
668                     .map(o -> " || (" + o + ")").collect(Collectors.joining());
669             }
670             String before = s.substring(0, start);
671             String after = s.substring(end);
672             s = before + (before.isEmpty() || Character.isWhitespace(before.charAt(before.length() - 1)) ? "" : " ")
673                 + replacement + (after.isEmpty() || Character.isWhitespace(after.charAt(0)) ? "" : " ") + after;
674         }
675         return s;
676     }

723     /**
724      * Finds the start of the operand left of an &amp;.
725      *
726      * @param sql    SQL string
727      * @param quoted quoted positions, see {@link #quotedPositions(String)}
728      * @param amp    position of the operator
729      * @return start of the operand
730      */
731     private static int concatOperandStart(String sql, boolean[] quoted, int amp) {
732         int depth = 0;
733         int i = amp - 1;
734         while (i >= 0) {
735             char c = sql.charAt(i);
736             if (quoted[i]) {
737                 i--;
738             } else if (c == ')') {
739                 depth++;
740                 i--;
741             } else if (c == '(') {
742                 if (depth == 0) {
743                     return i + 1;
744                 }
745                 depth--;
746                 i--;
747             } else if (Character.isLetterOrDigit(c) || c == '_' || c == '$') {
748                 int wordEnd = i + 1;
749                 while (i >= 0 && !quoted[i] && (Character.isLetterOrDigit(sql.charAt(i)) || sql.charAt(i) == '_' || sql.charAt(i) == '$')) {
750                     i--;
751                 }
752                 if (depth == 0 && isConcatBoundary(sql.substring(i + 1, wordEnd))) {
753                     return wordEnd;
754                 }
755             } else if (depth == 0 && ",=<>;".indexOf(c) >= 0) {
756                 return i + 1;
757             } else {
758                 i--;
759             }
760         }
761         return 0;
762     }
