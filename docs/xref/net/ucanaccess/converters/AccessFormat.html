26  /**
27   * Formats values like the Access function Format, using the format language of VBA: named formats like "Standard" or
28   * "Long Date", up to four sections separated by semicolons, literals in quotes or after a backslash, the number
29   * placeholders 0 and #, the text placeholders @ and &amp; and the date tokens like dd, mmm, hh:nn or AM/PM.
30   * <p>
31   * Texts that come from the locale, like Yes and No or the names of months, are taken from Java and may differ from the
32   * texts Access shows in some languages.
33   *
34   * @since 5.3.1
35   */
36  final class AccessFormat {

47      /**
48       * Formats a value like the Access function Format.
49       *
50       * @param value           number, boolean, date, text or {@code null}
51       * @param format          format expression, {@code null} or empty for the general format
52       * @param firstDayOfWeek  first day of the week for the tokens w and ww, 1 for Sunday to 7 for Saturday, 0 for the
53       *                        default locale
54       * @param firstWeekOfYear first week of the year for the token ww, see DatePart
55       * @return formatted text, never {@code null}
56       * @throws UcanaccessSQLException if a week argument is out of range
57       */
58      static String format(Object value, String format, int firstDayOfWeek, int firstWeekOfYear) throws UcanaccessSQLException {
59          Object val = value instanceof Boolean ? (Boolean) value ? -1d : 0d
60              : value instanceof Number ? (Object) ((Number) value).doubleValue() : value;
61          if (format == null || format.isEmpty()) {
62              return val == null ? "" : general(val);
63          }
64          String named = formatNamed(val, format);
65          if (named != null) {
66              return named;
67          }
68          List<String> sections = splitSections(format);
69          if (isTextFormat(sections.get(0))) {
70              String text = val == null ? "" : val instanceof String ? (String) val : general(val);
71              return text.isEmpty() && sections.size() > 1 ? formatText("", sections.get(1)) : formatText(text, sections.get(0));
72          } else if (val == null) {
73              return sections.size() > 3 ? formatNumber(0, sections.get(3)) : "";
74          }
75          boolean dateFormat = !isNumberFormat(sections) && hasDateTokens(sections.get(0));
76          if (val instanceof String) {
77              val = toNumberOrDate((String) val, dateFormat);
78              if (val instanceof String) {
79                  return (String) val;
80              }
81          }
82          if (val instanceof Timestamp) {
83              return dateFormat ? formatDate((Timestamp) val, sections.get(0), firstDayOfWeek, firstWeekOfYear)
84                  : formatNumber(toSerial((Timestamp) val), sections);
85          }
86          double d = (Double) val;
87          return dateFormat ? formatDate(Functions.cdate(d), sections.get(0), firstDayOfWeek, firstWeekOfYear)
88              : formatNumber(d, sections);
89      }

91      /**
92       * Converts a value to text without a format, like Format with one argument.
93       *
94       * @param value number, date or text
95       * @return text
96       * @throws UcanaccessSQLException never for valid values
97       */
98      private static String general(Object value) throws UcanaccessSQLException {
99          if (value instanceof Double) {
100             return generalNumber((Double) value);
101         } else if (value instanceof Timestamp) {
102             return formatNamedDate((Timestamp) value, GENERAL_DATE);
103         }
104         return value.toString();
105     }

107     /**
108      * Converts a number to text like Access with up to 15 significant digits. A number of 1E15 or more that does not
109      * fit into 15 digits is shown in scientific notation, as Access does, so 2^53 becomes 9.00719925474099E+15.
110      *
111      * @param d number
112      * @return text with the decimal separator of the default locale
113      */
114     static String generalNumber(double d) {
115         if (!Double.isFinite(d)) {
116             return Double.toString(d);
117         }
118         BigDecimal exact = BigDecimal.valueOf(d);
119         BigDecimal rounded = exact.round(new MathContext(15));
120         if (Math.abs(d) < 1E15 || rounded.compareTo(exact) == 0) {
121             return Functions.cstr(d);
122         }
123         int exponent = rounded.precision() - rounded.scale() - 1;
124         String mantissa = rounded.movePointLeft(exponent).stripTrailingZeros().toPlainString();
125         return String.format("%sE+%02d", mantissa.replace('.', decimalSeparator()), exponent);
126     }

128     /**
129      * Formats a value with a named format like "Standard" or "Short Date".
130      *
131      * @param value  number, date, text or {@code null}
132      * @param format format expression
133      * @return formatted text, or {@code null} if the format is not a named one
134      * @throws UcanaccessSQLException never for valid values
135      */
136     private static String formatNamed(Object value, String format) throws UcanaccessSQLException {
137         String name = format.trim().toLowerCase(Locale.ROOT);
138         boolean number = isNamedNumber(name);
139         if (!number && !isNamedDate(name)) {
140             return null;
141         } else if (value == null) {
142             return "";
143         }
144         Object val = value instanceof String ? toNumberOrDate((String) value, !number) : value;
145         if (val instanceof String) {
146             return (String) val;
147         } else if (number) {
148             return formatNamedNumber(val instanceof Timestamp ? toSerial((Timestamp) val) : (Double) val, name);
149         }
150         return formatNamedDate(val instanceof Double ? Functions.cdate((Double) val) : (Timestamp) val, name);
151     }

175     /**
176      * Formats a number with a named number format.
177      *
178      * @param d    number
179      * @param name format name in lower case
180      * @return formatted text
181      */
182     private static String formatNamedNumber(double d, String name) {
183         switch (name) {
184             case "general number":
185                 return generalNumber(d);
186             case "currency":
187                 return NumberFormat.getCurrencyInstance().format(d).replace(' ', ' ');
188             case "fixed":
189                 return formatNumber(d, List.of("0.00"));
190             case "standard":
191                 return formatNumber(d, List.of("#,##0.00"));
192             case "percent":
193                 return formatNumber(d, List.of("0.00%"));
194             case "scientific":
195                 return formatNumber(d, List.of("0.00E+00"));
196             case "yes/no":
197                 return d == 0 ? "No" : "Yes";
198             case "true/false":
199                 return d == 0 ? "False" : "True";
200             default:
201                 return d == 0 ? "Off" : "On";
202         }
203     }

205     /**
206      * Formats a date with a named date format, taking the patterns from the regional settings.
207      *
208      * @param t    date
209      * @param name format name in lower case
210      * @return formatted text
211      */
212     static String formatNamedDate(Timestamp t, String name) {
213         RegionalSettings reg = getRegionalSettings();
214         switch (name) {
215             case "long date":
216                 return formatPattern(t, reg.getLongDatePattern());
217             case "medium date":
218                 return formatPattern(t, reg.getMediumDatePattern());
219             case "short date":
220                 return formatPattern(t, reg.getShortDatePattern());
221             case "long time":
222                 return formatPattern(t, reg.getLongTimePattern());
223             case "medium time":
224                 return formatPattern(t, reg.getMediumTimePattern());
225             case "short time":
226                 return formatPattern(t, reg.getShortTimePattern());
227             default:
228                 // like Access, leave out a midnight time and the base date
229                 LocalDateTime ldt = t.toLocalDateTime();
230                 if (ldt.toLocalTime().equals(LocalTime.MIDNIGHT)) {
231                     return formatPattern(t, reg.getShortDatePattern());
232                 } else if (ldt.toLocalDate().equals(BASE_DATE.toLocalDate())) {
233                     return formatPattern(t, reg.getLongTimePattern());
234                 }
235                 return formatPattern(t, reg.getGeneralPattern());
236         }
237     }

239     /**
240      * Formats a date with a Java date pattern and the AM and PM texts of the regional settings.
241      *
242      * @param t       date
243      * @param pattern pattern of {@link SimpleDateFormat}
244      * @return formatted text
245      */
246     private static String formatPattern(Timestamp t, String pattern) {
247         RegionalSettings reg = getRegionalSettings();
248         String ret = Functions.createSimpleDateFormat(pattern).format(t);
249         if (!"true".equalsIgnoreCase(reg.getRS())) {
250             if (!"AM".equals(reg.getAM())) {
251                 ret = ret.replace("AM", reg.getAM());
252             }
253             if (!"PM".equals(reg.getPM())) {
254                 ret = ret.replace("PM", reg.getPM());
255             }
256         } else {
257             ret = ret.replace(reg.getPM(), "PM").replace(reg.getAM(), "AM");
258         }
259         return ret;
260     }

262     /**
263      * Converts a text to a number or a date, as Format does before it applies a number or date format. A text like
264      * 5.3.2026 can be both in some locales, so a date format tries the date first.
265      *
266      * @param s          text
267      * @param preferDate whether to try the date first
268      * @return number, date, or the text itself if it is neither
269      */
270     private static Object toNumberOrDate(String s, boolean preferDate) {
271         if (s.isEmpty()) {
272             return s;
273         }
274         Timestamp t = preferDate ? Functions.dateValue(s, false) : null;
275         if (t != null) {
276             return t;
277         } else if (Functions.isNumeric(s)) {
278             Number n = NumberFormat.getInstance().parse(s, new ParsePosition(0));
279             if (n != null) {
280                 return n.doubleValue();
281             }
282         }
283         t = preferDate ? null : Functions.dateValue(s, false);
284         return t == null ? s : t;
285     }

287     /**
288      * Converts a date to the Access date value, the days since 1899-12-30 with the time as fraction. As in Access, the
289      * fraction of a date before 1899-12-30 counts in the negative direction.
290      *
291      * @param t date
292      * @return date value
293      */
294     static double toSerial(Timestamp t) {
295         LocalDateTime ldt = t.toLocalDateTime();
296         long days = ChronoUnit.DAYS.between(BASE_DATE.toLocalDate(), ldt.toLocalDate());
297         double time = ldt.toLocalTime().toNanoOfDay() / 86_400_000_000_000d;
298         return days < 0 ? days - time : days + time;
299     }

301     /**
302      * Splits a format expression into its sections at semicolons outside literals.
303      *
304      * @param format format expression
305      * @return sections, at least one
306      */
307     private static List<String> splitSections(String format) {
308         List<String> sections = new ArrayList<>();
309         StringBuilder sb = new StringBuilder();
310         for (int i = 0; i < format.length(); i++) {
311             char c = format.charAt(i);
312             if (c == ';') {
313                 sections.add(sb.toString());
314                 sb.setLength(0);
315                 continue;
316             }
317             sb.append(c);
318             if (c == '\\' && i + 1 < format.length()) {
319                 sb.append(format.charAt(++i));
320             } else if (c == '"') {
321                 int end = format.indexOf('"', i + 1);
322                 end = end < 0 ? format.length() - 1 : end;
323                 sb.append(format, i + 1, end + 1);
324                 i = end;
325             }
326         }
327         sections.add(sb.toString());
328         return sections;
329     }

373     /**
374      * Formats a number with the sections of a number format: the first for positive numbers, the second for negative
375      * numbers and the third for zero; an empty or missing section falls back to the first one. A number that rounds to
376      * zero is zero, so it gets the zero section and no minus sign.
377      *
378      * @param d        number
379      * @param sections format sections
380      * @return formatted text
381      */
382     private static String formatNumber(double d, List<String> sections) {
383         String negative = sections.size() > 1 && !sections.get(1).isEmpty() ? sections.get(1) : null;
384         String zero = sections.size() > 2 && !sections.get(2).isEmpty() ? sections.get(2) : null;
385         String section = d < 0 && negative != null ? negative : sections.get(0);
386         NumberSection ns = new NumberSection(section);
387         if (ns.isZero(Math.abs(d))) {
388             return zero == null ? ns.format(0) : formatNumber(0, zero);
389         }
390         String text = ns.format(Math.abs(d));
391         return d < 0 && negative == null ? "-" + text : text;
392     }

405     /**
406      * Formats a text with a text section: @ stands for a character or a space, &amp; for a character or nothing. The
407      * placeholders are filled from the right, with ! from the left; &lt; and &gt; convert to lower or upper case.
408      *
409      * @param text    text
410      * @param section format section
411      * @return formatted text
412      */
413     private static String formatText(String text, String section) {
414         List<String> tokens = new ArrayList<>();
415         boolean leftToRight = false;
416         String s = text;
417         for (int i = 0; i < section.length(); i++) {
418             char c = section.charAt(i);
419             if (c == '@' || c == '&') {
420                 tokens.add(String.valueOf(c));
421             } else if (c == '<') {
422                 s = text.toLowerCase(Locale.getDefault());
423             } else if (c == '>') {
424                 s = text.toUpperCase(Locale.getDefault());
425             } else if (c == '!') {
426                 leftToRight = true;
427             } else {
428                 i = addLiteral(section, i, tokens);
429             }
430         }
431         int placeholders = (int) tokens.stream().filter(t -> "@".equals(t) || "&".equals(t)).count();
432         if (placeholders == 0) {
433             return literals(tokens) + s;
434         }
435         // the characters for each placeholder; without ! the last placeholder takes the excess, with ! the excess is cut
436         List<String> chars = new ArrayList<>();
437         int len = s.length();
438         for (int p = 0; p < placeholders; p++) {
439             int index = leftToRight ? p + Math.max(0, len - placeholders) : p - Math.max(0, placeholders - len);
440             if (index < 0 || index >= len) {
441                 chars.add(null);
442             } else {
443                 chars.add(!leftToRight && p == placeholders - 1 ? s.substring(index) : s.substring(index, index + 1));
444             }
445         }
446         StringBuilder sb = new StringBuilder();
447         int p = 0;
448         for (String t : tokens) {
449             if ("@".equals(t) || "&".equals(t)) {
450                 String ch = chars.get(p++);
451                 sb.append(ch != null ? ch : "@".equals(t) ? " " : "");
452             } else {
453                 sb.append(t.substring(1));
454             }
455         }
456         return sb.toString();
457     }

459     /**
460      * Adds the literal at a position of a section to a token list: text in quotes, a character after a backslash or
461      * the character itself; color names in brackets are skipped.
462      *
463      * @param section format section
464      * @param i       position of the literal
465      * @param tokens  token list to add to
466      * @return position of the last character of the literal
467      */
468     private static int addLiteral(String section, int i, List<String> tokens) {
469         char c = section.charAt(i);
470         if (c == '"') {
471             int end = section.indexOf('"', i + 1);
472             end = end < 0 ? section.length() : end;
473             tokens.add(LITERAL + section.substring(i + 1, end));
474             return end;
475         } else if (c == '\\') {
476             if (i + 1 < section.length()) {
477                 tokens.add(LITERAL + section.substring(i + 1, i + 2));
478             }
479             return i + 1;
480         } else if (c == '[') {
481             int end = section.indexOf(']', i);
482             return end < 0 ? section.length() : end;
483         }
484         tokens.add(LITERAL + c);
485         return i;
486     }

488     /**
489      * Joins the literals of a token list and leaves out all other tokens.
490      *
491      * @param tokens tokens and literals
492      * @return literal text
493      */
494     private static String literals(List<String> tokens) {
495         return tokens.stream().filter(t -> t.startsWith(LITERAL)).map(t -> t.substring(1)).collect(Collectors.joining());
496     }

498     /**
499      * Formats a date with a date section.
500      *
501      * @param t               date
502      * @param section         format section
503      * @param firstDayOfWeek  first day of the week, see {@link #format(Object, String, int, int)}
504      * @param firstWeekOfYear first week of the year, see {@link #format(Object, String, int, int)}
505      * @return formatted text
506      * @throws UcanaccessSQLException if a week argument is out of range
507      */
508     private static String formatDate(Timestamp t, String section, int firstDayOfWeek, int firstWeekOfYear)
509         throws UcanaccessSQLException {
510         List<String> tokens = tokenizeDate(section);
511         boolean twelveHours = tokens.stream().anyMatch(tk -> tk.matches("(?i)@(AM/PM|A/P|AMPM)"));
512         LocalDateTime ldt = t.toLocalDateTime();
513         DateFormatSymbols symbols = DateFormatSymbols.getInstance();
514         StringBuilder sb = new StringBuilder();
515         String previous = "";
516         for (String tk : tokens) {
517             if (!tk.startsWith("@")) {
518                 sb.append(tk.substring(1));
519                 continue;
520             }
521             String token = tk.substring(1);
522             String lower = token.toLowerCase(Locale.ROOT);
523             if (lower.startsWith("m") && lower.length() <= 2 && previous.startsWith("h")) {
524                 lower = lower.replace('m', 'n');
525             }
526             sb.append(formatDateToken(ldt, t, token, lower, twelveHours, symbols, firstDayOfWeek, firstWeekOfYear));
527             previous = lower;
528         }
529         return sb.toString();
530     }

532     /**
533      * Formats one date token.
534      *
535      * @param ldt             date
536      * @param t               the same date as timestamp
537      * @param token           token as written in the format
538      * @param lower           token in lower case, with a minute written as m already turned into n
539      * @param twelveHours     whether hours count from 1 to 12, as the format has AM/PM
540      * @param symbols         names of months and days
541      * @param firstDayOfWeek  first day of the week, see {@link #format(Object, String, int, int)}
542      * @param firstWeekOfYear first week of the year, see {@link #format(Object, String, int, int)}
543      * @return formatted token
544      * @throws UcanaccessSQLException if a week argument is out of range
545      */
546     private static String formatDateToken(LocalDateTime ldt, Timestamp t, String token, String lower, boolean twelveHours,
547         DateFormatSymbols symbols, int firstDayOfWeek, int firstWeekOfYear) throws UcanaccessSQLException {
548         int hour = twelveHours ? (ldt.getHour() + 11) % 12 + 1 : ldt.getHour();
549         RegionalSettings reg = getRegionalSettings();
550         switch (lower) {
551             case "c":
552                 return formatNamedDate(t, GENERAL_DATE);
553             case "d":
554                 return String.valueOf(ldt.getDayOfMonth());
555             case "dd":
556                 return String.format("%02d", ldt.getDayOfMonth());
557             case "ddd":
558                 return symbols.getShortWeekdays()[ldt.getDayOfWeek().getValue() % 7 + 1].replace(".", "");
559             case "dddd":
560                 return symbols.getWeekdays()[ldt.getDayOfWeek().getValue() % 7 + 1];
561             case "ddddd":
562                 return formatPattern(t, reg.getShortDatePattern());
563             case "dddddd":
564                 return formatPattern(t, reg.getLongDatePattern());
565             case "w":
566                 return String.valueOf(weekday(ldt.toLocalDate(), firstDayOfWeek));
567             case "ww":
568                 return String.valueOf(Functions.weekOfYear(ldt.toLocalDate(), firstDayOfWeek, firstWeekOfYear));
569             case "m":
570                 return String.valueOf(ldt.getMonthValue());
571             case "mm":
572                 return String.format("%02d", ldt.getMonthValue());
573             case "mmm":
574                 return symbols.getShortMonths()[ldt.getMonthValue() - 1].replace(".", "");
575             case "mmmm":
576                 return symbols.getMonths()[ldt.getMonthValue() - 1];
577             case "q":
578                 return String.valueOf((ldt.getMonthValue() + 2) / 3);
579             case "y":
580                 return String.valueOf(ldt.getDayOfYear());
581             case "yy":
582                 return String.format("%02d", ldt.getYear() % 100);
583             case "yyyy":
584                 return String.valueOf(ldt.getYear());
585             case "h":
586                 return String.valueOf(hour);
587             case "hh":
588                 return String.format("%02d", hour);
589             case "n":
590                 return String.valueOf(ldt.getMinute());
591             case "nn":
592                 return String.format("%02d", ldt.getMinute());
593             case "s":
594                 return String.valueOf(ldt.getSecond());
595             case "ss":
596                 return String.format("%02d", ldt.getSecond());
597             case "ttttt":
598                 return formatPattern(t, reg.getLongTimePattern());
599             case "ampm":
600                 return ldt.getHour() < 12 ? reg.getAM() : reg.getPM();
601             default:
602                 // AM/PM or A/P in the case written in the format
603                 String[] parts = token.split("/");
604                 return ldt.getHour() < 12 ? parts[0] : parts[1];
605         }
606     }

608     /**
609      * Returns the day of the week from 1 to 7, counted from the given first day.
610      *
611      * @param date           date
612      * @param firstDayOfWeek 1 for Sunday to 7 for Saturday, 0 for the default locale
613      * @return day of the week
614      */
615     private static int weekday(LocalDate date, int firstDayOfWeek) {
616         int first = firstDayOfWeek == 0 ? WeekFields.of(Locale.getDefault()).getFirstDayOfWeek().getValue() % 7 + 1
617             : firstDayOfWeek;
618         int sundayFirst = date.getDayOfWeek().getValue() % 7 + 1;
619         return (sundayFirst - first + 7) % 7 + 1;
620     }

622     /**
623      * Splits a date section into tokens. A token starts with @, followed by the token as written; a run of the same
624      * letter longer than the longest token is split, so mmmmm is mmmm and m. Other entries are literal text, with / and
625      * : replaced by the date and time separators of the locale.
626      *
627      * @param section format section
628      * @return tokens and literals
629      */
630     private static List<String> tokenizeDate(String section) {
631         List<String> tokens = new ArrayList<>();
632         String dateSeparator = separator(getRegionalSettings().getShortDatePattern(), "/");
633         String timeSeparator = separator(getRegionalSettings().getLongTimePattern(), ":");
634         for (int i = 0; i < section.length(); i++) {
635             char c = section.charAt(i);
636             char lower = Character.toLowerCase(c);
637             int max = maxTokenLength(lower);
638             if (section.regionMatches(true, i, "AM/PM", 0, 5)) {
639                 tokens.add("@" + section.substring(i, i + 5));
640                 i += 4;
641             } else if (section.regionMatches(true, i, "A/P", 0, 3)) {
642                 tokens.add("@" + section.substring(i, i + 3));
643                 i += 2;
644             } else if (section.regionMatches(true, i, "AMPM", 0, 4)) {
645                 tokens.add("@" + section.substring(i, i + 4));
646                 i += 3;
647             } else if (lower == 't') {
648                 int run = runLength(section, i);
649                 tokens.add(run >= 5 ? "@ttttt" : LITERAL + section.substring(i, i + run));
650                 i += (run >= 5 ? 5 : run) - 1;
651             } else if (max > 0) {
652                 int run = Math.min(runLength(section, i), max);
653                 if (lower == 'y' && run == 3) {
654                     run = 2;
655                 }
656                 tokens.add("@" + section.substring(i, i + run));
657                 i += run - 1;
658             } else if (c == '/') {
659                 tokens.add(LITERAL + dateSeparator);
660             } else if (c == ':') {
661                 tokens.add(LITERAL + timeSeparator);
662             } else {
663                 i = addLiteral(section, i, tokens);
664             }
665         }
666         return tokens;
667     }
668 
669     /**
670      * Returns the longest token for a letter of a date format.
671      *
672      * @param lower letter in lower case
673      * @return length of the longest token, 0 if the letter is no token
674      */
675     private static int maxTokenLength(char lower) {
676         switch (lower) {
677             case 'c':
678             case 'q':
679                 return 1;
680             case 'd':
681                 return 6;
682             case 'm':
683             case 'y':
684                 return 4;
685             case 'w':
686             case 'h':
687             case 'n':
688             case 's':
689                 return 2;
690             default:
691                 return 0;
692         }
693     }

695     /**
696      * Counts how often the character at a position repeats, ignoring case.
697      *
698      * @param s text
699      * @param i position
700      * @return length of the run, at least 1
701      */
702     private static int runLength(String s, int i) {
703         char c = Character.toLowerCase(s.charAt(i));
704         int end = i + 1;
705         while (end < s.length() && Character.toLowerCase(s.charAt(end)) == c) {
706             end++;
707         }
708         return end - i;
709     }

711     /**
712      * Returns the separator a Java date pattern uses, that is its first character that is no letter.
713      *
714      * @param pattern      pattern of {@link SimpleDateFormat}
715      * @param defaultValue separator if the pattern has none
716      * @return separator
717      */
718     private static String separator(String pattern, String defaultValue) {
719         for (int i = 0; i < pattern.length(); i++) {
720             char c = pattern.charAt(i);
721             if (!Character.isLetter(c) && c != '\'' && c != ' ') {
722                 return String.valueOf(c);
723             }
724         }
725         return defaultValue;
726     }

737     /**
738      * One section of a number format, split into integer part, fraction and exponent.
739      */
740     private static final class NumberSection {
741         private final List<String> integerTokens  = new ArrayList<>();
742         private final List<String> fractionTokens = new ArrayList<>();
743         private final List<String> exponentTokens = new ArrayList<>();
744 
745         private boolean            point;
746         private boolean            grouping;
747         private boolean            percent;
748         private int                scale;
749         private String             exponent;
750         private int                fractionDigits;
751         private int                integerDigits;
752 
753         /**
754          * Parses a number section.
755          *
756          * @param section format section
757          */
758         NumberSection(String section) {
759             List<String> part = integerTokens;
760             for (int i = 0; i < section.length(); i++) {
761                 char c = section.charAt(i);
762                 boolean inInteger = !point && exponent == null;
763                 if (c == '0' || c == '#') {
764                     part.add(String.valueOf(c));
765                     if (inInteger) {
766                         integerDigits++;
767                     } else if (exponent == null) {
768                         fractionDigits++;
769                     }
770                 } else if (c == '.' && inInteger) {
771                     point = true;
772                     part = fractionTokens;
773                 } else if (c == ',') {
774                     if (inInteger) {
775                         part.add(",");
776                     }
777                 } else if ((c == 'E' || c == 'e') && i + 1 < section.length() && "+-".indexOf(section.charAt(i + 1)) >= 0
778                     && exponent == null) {
779                     exponent = section.substring(i, i + 2);
780                     part = exponentTokens;
781                     i++;
782                 } else if (c == '%') {
783                     percent = true;
784                     part.add(LITERAL + "%");
785                 } else {
786                     i = addLiteral(section, i, part);
787                 }
788             }
789             resolveCommas();
790         }
791 
792         /**
793          * Turns the commas of the integer part into grouping, if a placeholder comes before and after, or into
794          * scaling by 1000, if they come after the last placeholder.
795          */
796         private void resolveCommas() {
797             int last = -1;
798             for (int i = 0; i < integerTokens.size(); i++) {
799                 if (isPlaceholder(integerTokens.get(i))) {
800                     last = i;
801                 }
802             }
803             boolean seenPlaceholder = false;
804             for (int i = 0; i < integerTokens.size(); i++) {
805                 String t = integerTokens.get(i);
806                 if (isPlaceholder(t)) {
807                     seenPlaceholder = true;
808                 } else if (",".equals(t)) {
809                     if (i > last) {
810                         scale++;
811                     } else if (seenPlaceholder) {
812                         grouping = true;
813                     }
814                     integerTokens.remove(i--);
815                     last--;
816                 }
817             }
818         }
819 
820         /**
821          * Tells whether a token is a digit placeholder.
822          *
823          * @param t token
824          * @return {@code true} for 0 or #
825          */
826         private static boolean isPlaceholder(String t) {
827             return "0".equals(t) || "#".equals(t);
828         }
829 
830         /**
831          * Scales a non-negative number by the percent sign and the scaling commas and rounds it to 15 digits.
832          *
833          * @param d number
834          * @return scaled number
835          */
836         private BigDecimal scaled(double d) {
837             BigDecimal bd = BigDecimal.valueOf(d).round(new MathContext(15));
838             if (percent) {
839                 bd = bd.movePointRight(2);
840             }
841             return bd.movePointLeft(3 * scale);
842         }
843 
844         /**
845          * Tells whether a non-negative number shows as zero in this section.
846          *
847          * @param d number
848          * @return {@code true} if the number rounds to zero
849          */
850         boolean isZero(double d) {
851             return exponent == null ? scaled(d).setScale(fractionDigits, RoundingMode.HALF_UP).signum() == 0 : d == 0;
852         }
853 
854         /**
855          * Formats a non-negative number.
856          *
857          * @param d number
858          * @return formatted text
859          */
860         String format(double d) {
861             if (integerDigits + fractionDigits == 0) {
862                 return literals(integerTokens) + literals(fractionTokens) + literals(exponentTokens);
863             }
864             BigDecimal bd = scaled(d);
865             int exp = 0;
866             if (exponent != null && bd.signum() != 0) {
867                 int digits = Math.max(1, integerDigits);
868                 exp = bd.precision() - bd.scale() - digits;
869                 BigDecimal mantissa = bd.movePointLeft(exp).setScale(fractionDigits, RoundingMode.HALF_UP);
870                 if (mantissa.precision() - mantissa.scale() > digits) {
871                     exp++;
872                 }
873                 bd = bd.movePointLeft(exp);
874             }
875             bd = bd.setScale(fractionDigits, RoundingMode.HALF_UP);
876             String plain = bd.toPlainString();
877             String intDigits = plain.contains(".") ? plain.substring(0, plain.indexOf('.')) : plain;
878             String fraction = plain.contains(".") ? plain.substring(plain.indexOf('.') + 1) : "";
879             StringBuilder sb = new StringBuilder(formatInteger("0".equals(intDigits) ? "" : intDigits));
880             if (point) {
881                 sb.append(decimalSeparator());
882             }
883             sb.append(formatFraction(fraction));
884             if (exponent != null) {
885                 sb.append(formatExponent(exp));
886             }
887             return sb.toString();
888         }
889 
890         /**
891          * Distributes the integer digits over the placeholders from the right; excess digits go to the first one.
892          *
893          * @param digits integer digits without leading zeros, empty for zero
894          * @return formatted integer part
895          */
896         private String formatInteger(String digits) {
897             List<String> out = new ArrayList<>();
898             List<Boolean> isDigit = new ArrayList<>();
899             int di = digits.length() - 1;
900             int first = -1;
901             for (int i = 0; i < integerTokens.size(); i++) {
902                 if (isPlaceholder(integerTokens.get(i)) && first < 0) {
903                     first = i;
904                 }
905             }
906             for (int i = integerTokens.size() - 1; i >= 0; i--) {
907                 String t = integerTokens.get(i);
908                 if (!isPlaceholder(t)) {
909                     out.add(0, t.substring(1));
910                     isDigit.add(0, false);
911                     continue;
912                 }
913                 if (di >= 0) {
914                     out.add(0, String.valueOf(digits.charAt(di--)));
915                     isDigit.add(0, true);
916                 } else if ("0".equals(t)) {
917                     out.add(0, "0");
918                     isDigit.add(0, true);
919                 }
920                 if (i == first) {
921                     while (di >= 0) {
922                         out.add(0, String.valueOf(digits.charAt(di--)));
923                         isDigit.add(0, true);
924                     }
925                 }
926             }
927             if (first < 0) {
928                 out.add(digits);
929                 isDigit.add(false);
930             }
931             StringBuilder sb = new StringBuilder();
932             long total = isDigit.stream().filter(b -> b).count();
933             long remaining = total;
934             char groupingSeparator = DecimalFormatSymbols.getInstance().getGroupingSeparator();
935             for (int i = 0; i < out.size(); i++) {
936                 if (isDigit.get(i)) {
937                     if (grouping && remaining < total && remaining % 3 == 0) {
938                         sb.append(groupingSeparator);
939                     }
940                     remaining--;
941                 }
942                 sb.append(out.get(i));
943             }
944             return sb.toString();
945         }
946 
947         /**
948          * Puts the fraction digits into the placeholders; trailing zeros in # placeholders are left out.
949          *
950          * @param digits fraction digits, as many as placeholders
951          * @return formatted fraction
952          */
953         private String formatFraction(String digits) {
954             List<String> placeholders = fractionTokens.stream().filter(NumberSection::isPlaceholder).collect(Collectors.toList());
955             int keep = digits.length();
956             while (keep > 0 && "#".equals(placeholders.get(keep - 1)) && digits.charAt(keep - 1) == '0') {
957                 keep--;
958             }
959             StringBuilder sb = new StringBuilder();
960             int di = 0;
961             for (String t : fractionTokens) {
962                 if (isPlaceholder(t)) {
963                     if (di < keep) {
964                         sb.append(digits.charAt(di));
965                     }
966                     di++;
967                 } else {
968                     sb.append(t.substring(1));
969                 }
970             }
971             return sb.toString();
972         }
973 
974         /**
975          * Formats the exponent: E+ shows the sign always, E- only for a negative exponent.
976          *
977          * @param exp exponent
978          * @return formatted exponent with the literals after it
979          */
980         private String formatExponent(int exp) {
981             int minDigits = (int) exponentTokens.stream().filter("0"::equals).count();
982             StringBuilder sb = new StringBuilder().append(exponent.charAt(0));
983             if (exp < 0 || exponent.charAt(1) == '+') {
984                 sb.append(exp < 0 ? '-' : '+');
985             }
986             String digits = String.valueOf(Math.abs(exp));
987             if (exp == 0 && minDigits == 0) {
988                 digits = "";
989             }
990             sb.append("0".repeat(Math.max(0, minDigits - digits.length()))).append(digits);
991             sb.append(literals(exponentTokens));
992             return sb.toString();
993         }
994     }
