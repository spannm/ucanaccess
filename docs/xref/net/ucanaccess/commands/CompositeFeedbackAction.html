1   package net.ucanaccess.commands;
2   
3   import java.sql.SQLException;
4   import java.util.ArrayList;
5   import java.util.List;
6   
7   /**
8    * Composite pattern implementation for {@link IFeedbackAction}.
9    * <p>
10   * Holds a list of feedback actions to be executed sequentially.
11   */
12  public class CompositeFeedbackAction implements IFeedbackAction {
13      private final List<IFeedbackAction> actions = new ArrayList<>();
14  
15      @Override
16      public void doAction(ICommand toChange) throws SQLException {
17          for (IFeedbackAction action : actions) {
18              action.doAction(toChange);
19          }
20      }
21  
22      /**
23       * Adds a feedback action to the composite.
24       *
25       * @param ifa the action to add
26       * @return true if the action was added, false otherwise
27       */
28      public boolean add(IFeedbackAction ifa) {
29          return ifa != null && actions.add(ifa);
30      }
31  
32      @Override
33      public String toString() {
34          return String.format("%s[actions=%s]", getClass().getSimpleName(), actions);
35      }
36  
37  }
