7    * <p>
8    * This is primarily used to synchronize the state of the HSQLDB mirror with the
9    * results of the Jackcess persistence layer (e.g., updating autoincrement values
10   * or handling complex type identifiers).
11   */
12  public interface IFeedbackAction {
13  
14      /**
15       * Executes the feedback action on the specified command.
16       * <p>
17       * This method is called after the physical persistence to the Access file
18       * has succeeded, allowing for final adjustments to the command's state.
19       *
20       * @param toChange the command instance that may be modified by this action
21       * @throws SQLException if an error occurs during the execution of the feedback action
22       */
23      void doAction(ICommand toChange) throws SQLException;
24  
25  }
