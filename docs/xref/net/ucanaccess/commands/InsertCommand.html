64      private Object[] mementoRow() {
65          Object[] clone = new Object[newRow.length];
66          int i = 0;
67          for (Object obj : newRow) {
68              clone[i] = obj;
69              ++i;
70          }
71          return clone;
72      }

74      private void initComplex() {
75          for (int i = 0; i < newRow.length; ++i) {
76              if (newRow[i] instanceof ComplexBase) {
77                  newRow[i] = Column.AUTO_NUMBER;
78              }
79          }
80      }

82      /**
83       * Adds the new row to the given table.
84       *
85       * @return the table the row was added to and the id of the new row
86       */
87      private Map.Entry<Table, RowId> insertRow(Table tbl) throws IOException {
88          // the element after the last column receives the id of the new row
89          int numCols = tbl.getColumnCount();
90          Object[] row = Arrays.copyOf(newRow, numCols + 1);
91          row[numCols] = ColumnImpl.RETURN_ROW_ID;
92          try {
93              tbl.addRow(row);
94          } catch (ConstraintViolationException ex) {
95              List<? extends Column> lc = tbl.getColumns();
96              boolean retry = false;
97              for (Column col : lc) {
98                  if (col.isAutoNumber()) {
99                      retry = true;
100                     break;
101                 }
102             }
103             if (!retry) {
104                 throw ex;
105             }
106             Database db = tbl.getDatabase();
107             File fl = db.getFile();
108             DBReferenceSingleton dbsin = DBReferenceSingleton.getInstance();
109             DBReference ref = dbsin.getReference(fl);
110             ref.reloadDbIO();
111             dbIO = ref.getDbIO();
112             tbl = dbIO.getTable(tableName);
113             tbl.addRow(row);
114         }
115         // the table fills in generated values such as autonumbers and column defaults
116         System.arraycopy(row, 0, newRow, 0, Math.min(newRow.length, numCols));
117         return new SimpleImmutableEntry<>(tbl, row[numCols] instanceof RowId ? (RowId) row[numCols] : null);
118     }

120     /**
121      * Jackcess replaces {@code null} with the column default on insert. The row has already been inserted into HSQLDB,
122      * which applied the defaults of omitted columns, so a remaining {@code null} was set explicitly and is restored.
123      */
124     private void restoreExplicitNulls(Table tbl, RowId rowId, Object[] memento, List<? extends Column> colList)
125         throws IOException, SQLException {
126         if (rowId == null) {
127             return;
128         }
129         Cursor cursor = null;
130         int j = 0;
131         for (Column col : colList) {
132             if (memento[j] == null && newRow[j] != null && !col.isAutoNumber() && col.getType() != DataType.BOOLEAN
133                 && col.getProperties().getValue(PropertyMap.DEFAULT_VALUE_PROP) != null && hasNativeDefault(col)) {
134                 if (cursor == null) {
135                     cursor = CursorBuilder.createCursor(tbl);
136                     cursor.findRow(rowId);
137                 }
138                 cursor.setCurrentRowValue(col, null);
139                 newRow[j] = null;
140             }
141             j++;
142         }
143     }

145     /**
146      * Only a native HSQLDB column default leaves an explicit {@code null} in the row. Defaults that UCanAccess could not
147      * translate are left to Jackcess.
148      */
149     private static boolean hasNativeDefault(Column col) throws SQLException {
150         UcanaccessConnection conn = UcanaccessConnection.getCtxConnection();
151         return conn != null && new Metadata(conn.getHSQLDBConnection()).hasNativeColumnDefault(col.getTable().getName(), col.getName());
152     }

154     @Override
155     public IFeedbackAction persist() throws SQLException {
156         try {
157             AutoNumberAction ana = null;
158             if (table == null) {
159                 table = dbIO.getTable(tableName);
160             }
161             Object[] memento = mementoRow();
162             initComplex();
163             int j = 0;
164             List<? extends Column> colList = table.getColumns();
165             if (table.getDatabase().getColumnOrder().equals(ColumnOrder.DISPLAY)) {
166                 Object[] newRowReorded = new Object[newRow.length];
167                 Column[] cllReorded = new Column[newRow.length];
168                 for (Column col : table.getColumns()) {
169                     newRowReorded[col.getColumnIndex()] = newRow[j];
170                     memento[col.getColumnIndex()] = newRow[j];
171                     cllReorded[col.getColumnIndex()] = col;
172                     j++;
173                 }
174                 newRow = newRowReorded;
175                 colList = Arrays.asList(cllReorded);
176             }
177 
178             Map.Entry<Table, RowId> inserted = insertRow(table);
179             restoreExplicitNulls(inserted.getKey(), inserted.getValue(), memento, colList);
180             j = 0;
181             for (Column col : colList) {
182                 ColumnImpl colImpl = (ColumnImpl) col;
183                 if (colImpl.isAutoNumber() && !memento[j].equals(newRow[j])
184                         && !colImpl.getAutoNumberGenerator().getType().equals(DataType.COMPLEX_TYPE)) {
185 
186                     if (colImpl.getAutoNumberGenerator().getType().equals(DataType.LONG)) {
187                         AutoNumberManager.reset(colImpl, (Integer) newRow[j]);
188                     }
189                     ana = new AutoNumberAction(table, memento, newRow);
190                 }
191 
192                 if (colImpl.getType() == DataType.COMPLEX_TYPE) {
193                     ComplexValueForeignKey rowFk = (ComplexValueForeignKey) colImpl.getRowValue(newRow);
194                     if (memento[j] instanceof Attachment[]) {
195                         Attachment[] atcs = (Attachment[]) memento[j];
196                         for (Attachment atc : atcs) {
197                             rowFk.addAttachment(atc.getUrl(), atc.getName(), atc.getType(), atc.getData(),
198                                     atc.getTimeStamp(), atc.getFlags());
199 
200                         }
201                     } else if (memento[j] instanceof SingleValue[]) {
202                         SingleValue[] vs = (SingleValue[]) memento[j];
203                         for (SingleValue v : vs) {
204                             rowFk.addMultiValue(v.getValue());
205                         }
206 
207                     } else if (memento[j] instanceof Version[]) {
208                         Version[] vs = (Version[]) memento[j];
209                         for (Version v : vs) {
210                             rowFk.addVersion(v.getValue(), v.getModifiedDate());
211                         }
212                     }
213                 }
214                 ++j;
215             }
216             BlobAction ba = new BlobAction(table, newRow);
217             ba.doAction(this);
218             return ana;
219         } catch (IOException ex) {
220             throw new UcanaccessSQLException(ex);
221         }
222     }

224     @Override
225     public IFeedbackAction rollback() throws SQLException {
226         if (table != null) {
227             DeleteCommand dc = new DeleteCommand(table, new Persist2Jet().getRowPattern(newRow, table),
228                     execId);
229             return dc.persist();
230         } else {
231             // a drop table cleans all
232             return null;
233         }
234     }
