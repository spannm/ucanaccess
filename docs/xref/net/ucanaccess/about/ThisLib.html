15  /**
16   * Provides metadata and build information about this library when executed directly via CLI.
17   */
18  public final class ThisLib {
19  
20      private ThisLib() {
21      }

23      /**
24       * Main entry point when the JAR file is executed directly.
25       * <p>
26       * Reads the {@code MANIFEST.MF} file from the JAR package and prints formatted metadata
27       * to {@code System.out}.
28       *
29       * @param args command-line arguments (unused)
30       */
31      @SuppressWarnings("java:S106") // prints the build information to the console by design
32      public static void main(String[] args) {
33          readManifest().ifPresent(m -> System.out.print(buildInfo(m)));
34      }
35  
36      /**
37       * Builds the formatted metadata text printed when the JAR file is executed directly.
38       *
39       * @param manifest the manifest to read metadata from, may be {@code null}
40       * @return the formatted metadata text, terminated with a trailing line separator
41       */
42      static String buildInfo(Manifest manifest) {
43          String title = getAttribute(manifest, "Implementation-Title", "Project-Name").orElse("Java Library");
44          String version = getAttribute(manifest, "Implementation-Version", "Project-Version").orElse("unknown");
45          String description = getAttribute(manifest, "Project-Description").orElse("");
46  
47          String gitCommit = getAttribute(manifest, "Git-Commit-Id", "X-BasePOM-Git-Commit-Id").orElse(null);
48  
49          String nl = System.lineSeparator();
50          StringBuilder sb = new StringBuilder(nl);
51          sb.append(title).append(" v").append(version).append(nl);
52  
53          if (!description.isEmpty()) {
54              sb.append(description).append(nl);
55          }
56          sb.append(nl);
57  
58          List<Map.Entry<String, String>> metadata = Arrays.asList(
59              entry("Vendor",      getAttribute(manifest, "Implementation-Vendor").orElse(null)),
60              entry("Build JDK",   getAttribute(manifest, "Build-Jdk-Spec").orElse(null)),
61              entry("Build Time",  getAttribute(manifest, "Build-Time").orElse(null)),
62              entry("Git Commit",  gitCommit),
63              entry("Git Branch",  getAttribute(manifest, "Git-Branch").orElse(null)),
64              entry("Homepage",    getAttribute(manifest, "Project-Url").orElse(null)),
65              entry("Issues",      getAttribute(manifest, "Issue-Management-Url").orElse(null)),
66              entry("Source Code", getAttribute(manifest, "Scm-Url").orElse(null)));
67  
68          for (Map.Entry<String, String> e : metadata) {
69              if (e.getValue() != null && !e.getValue().isEmpty()) {
70                  sb.append(String.format("%-15s: %s", e.getKey(), e.getValue())).append(nl);
71              }
72          }
73  
74          sb.append(nl);
75          sb.append("Please note: This JAR is a software library and not intended for direct CLI execution.")
76              .append(nl);
77  
78          return sb.toString();
79      }

81      /**
82       * Creates an immutable map entry representing a key-value metadata pair.
83       *
84       * @param key metadata key
85       * @param value metadata value
86       * @return immutable map entry
87       */
88      private static Map.Entry<String, String> entry(String key, String value) {
89          return new SimpleImmutableEntry<>(key, value);
90      }

92      /**
93       * Resolves and parses the {@code MANIFEST.MF} file from the location of this class's code source.
94       *
95       * @return an {@link Optional} containing the parsed {@link Manifest}, or empty if unavailable
96       */
97      private static Optional<Manifest> readManifest() {
98          CodeSource codeSource = ThisLib.class.getProtectionDomain().getCodeSource();
99          if (codeSource == null) {
100             return Optional.empty();
101         }
102 
103         URL location = codeSource.getLocation();
104         if (location == null) {
105             return Optional.empty();
106         }
107 
108         try {
109             URL manifestUrl = new URL("jar:" + location.toExternalForm() + "!/META-INF/MANIFEST.MF");
110             try (InputStream is = manifestUrl.openStream()) {
111                 return Optional.of(new Manifest(is));
112             }
113         } catch (IOException ignored) {
114             // running outside JAR context (e.g., IDE) is expected
115             return Optional.empty();
116         }
117     }
118 
119     /**
120      * Searches for the given attribute names in the main attributes first.
121      * <p>
122      * If not found, searches across all named individual sections in the manifest.
123      *
124      * @param manifest the manifest to search in, may be {@code null}
125      * @param attributeNames one or more attribute names to look for
126      * @return an {@link Optional} containing the first non-blank attribute value, or empty
127      */
128     private static Optional<String> getAttribute(Manifest manifest, String... attributeNames) {
129         if (manifest == null || attributeNames == null) {
130             return Optional.empty();
131         }
132 
133         for (String attrName : attributeNames) {
134             Optional<String> value = findInAttributes(manifest.getMainAttributes(), attrName);
135             if (value.isPresent()) {
136                 return value;
137             }
138 
139             for (Attributes sectionAttributes : manifest.getEntries().values()) {
140                 value = findInAttributes(sectionAttributes, attrName);
141                 if (value.isPresent()) {
142                     return value;
143                 }
144             }
145         }
146 
147         return Optional.empty();
148     }
149 
150     /**
151      * Extract and trims an attribute value from the given attributes object.
152      *
153      * @param attributes the attributes container, may be {@code null}
154      * @param name the name of the attribute
155      * @return an {@link Optional} containing the trimmed value, or empty if null or blank
156      */
157     private static Optional<String> findInAttributes(Attributes attributes, String name) {
158         if (attributes == null || name == null) {
159             return Optional.empty();
160         }
161         String val = attributes.getValue(name);
162         if (val == null) {
163             return Optional.empty();
164         }
165         String trimmed = val.trim();
166         return trimmed.isEmpty() ? Optional.empty() : Optional.of(trimmed);
167     }
